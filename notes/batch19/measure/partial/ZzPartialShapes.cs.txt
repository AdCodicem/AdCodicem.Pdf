    private static readonly StringBuilder Out = new();

    private sealed class W
    {
        private readonly MemoryStream _o = new();
        public readonly Dictionary<int, long> Offsets = [];
        public W(string version = "1.7") => T($"%PDF-{version}\n");
        public long P => _o.Position;
        public void T(string text) => _o.Write(Encoding.Latin1.GetBytes(text));
        public void B(byte[] b) => _o.Write(b);
        public void Obj(int n, string body) { Offsets[n] = P; T($"{n} 0 obj\n{body}\nendobj\n"); }
        public long Stream(int n, string entries, byte[] data) { var at = P; Offsets[n] = P; T($"{n} 0 obj\n<< {entries} /Length {data.Length} >>\nstream\n"); B(data); T("\nendstream\nendobj\n"); return at; }
        public void Pad(int count) { if (count > 0) T("%" + new string('-', count - 2) + "\n"); }
        public byte[] Bytes => _o.ToArray();
    }

    private static string Row(long offset, string kind = "n", long gen = 0) =>
        string.Create(CultureInfo.InvariantCulture, $"{offset:D10} {gen:D5} {kind}\r\n");

    private static byte[] Field(ulong value, int width, bool leadingOne = false)
    {
        var bytes = new byte[width];
        for (var i = 0; i < width; i++)
        {
            var shift = (width - 1 - i) * 8;
            bytes[i] = shift >= 64 ? (byte)(leadingOne && i == 0 ? 1 : 0) : (byte)(value >> shift);
        }

        return bytes;
    }

    private static byte[] SRows(int[] w, params (ulong T, ulong S, ulong G, int Lead)[] rows)
    {
        var d = new List<byte>();
        foreach (var (t, s, g, lead) in rows)
        {
            d.AddRange(Field(t, w[0], lead == 0));
            d.AddRange(Field(s, w[1], lead == 1));
            d.AddRange(Field(g, w[2], lead == 2));
        }

        return [.. d];
    }

    private static void Dump(string title, byte[] file, int[] objects)
    {
        Out.Append("### ").Append(title).Append(" (").Append(file.Length).Append(" bytes)\n");
        using (var document = PdfDocument.Open(file))
        {
            Out.Append("  opened: WasRepaired ").Append(document.WasRepaired).Append('\n');
            foreach (var s in document.Reader.Structure.Sections)
            {
                Out.Append(CultureInfo.InvariantCulture, $"  S {s.NamedBy} named={s.NamedOffset} offset={s.Offset} state={s.State} kind={s.Kind} fault={s.Fault ?? "-"} refused={s.RefusedRow ?? "-"} incomplete={s.Incomplete} trailerFault={s.TrailerFault} trailerAt={s.TrailerPosition} cut={s.CutByLimit} highest={s.HighestNumber}\n");
            }

            Out.Append("  at open: ").Append(document.Diagnostics.Count).Append(" diagnostic(s)\n");
            foreach (var d in document.Diagnostics) Out.Append("    D ").Append(d).Append('\n');
            var seen = document.Diagnostics.Count;
            foreach (var n in objects)
            {
                var value = document.GetObject(new PdfObjectId(n));
                var text = value.ToString() ?? string.Empty;
                Out.Append(CultureInfo.InvariantCulture, $"  GetObject({n}): {(text.Length > 60 ? text[..60] + "..." : text)}; WasRepaired: {document.WasRepaired}\n");
            }

            for (var i = seen; i < document.Diagnostics.Count; i++) Out.Append("    D+ ").Append(document.Diagnostics[i]).Append('\n');
        }

        using (var document = PdfDocument.Open(file))
        {
            var report = new PdfValidator().Validate(document);
            Out.Append("  PdfValidator: ").Append(report.Findings.Count).Append(" finding(s); WasRepaired ").Append(document.WasRepaired).Append('\n');
            foreach (var f in report.Findings) Out.Append("    F ").Append(f).Append('\n');
            foreach (var d in document.Diagnostics) Out.Append("    D(v) ").Append(d).Append('\n');
        }

        Out.Append('\n');
    }

    // ---- #207: a table the end of the file cuts before its trailer ----

    // The issue's shape: objects 1, 2, 4; a table indexing object 4 alone, /Size 5; object 3; the update's table naming
    // by /Prev the cut table at the end of the file.
    internal static byte[] Cut207Prev(string cutTail = null, int gap = 0)
    {
        var w = new W();
        w.Obj(1, "<< /Type /Catalog /Pages 2 0 R >>");
        w.Obj(2, "<< /Type /Pages /Kids [] /Count 0 >>");
        w.Obj(4, "(object 4)");
        w.T("xref\n4 1\n" + Row(w.Offsets[4]) + "trailer\n<< /Size 5 >>\n");
        w.Obj(3, "(object 3)");
        w.Pad(gap);
        var update = w.P;
        string Tail(long x) => "xref\n0 3\n" + Row(0, "f", 65535) + Row(w.Offsets[1]) + Row(w.Offsets[2]) +
            $"trailer\n<< /Size 5 /Root 1 0 R /Prev {x} >>\nstartxref\n{update}\n%%EOF\n";
        long cut = update + Tail(0).Length;
        while (update + Tail(cut).Length != cut) cut = update + Tail(cut).Length;
        w.T(Tail(cut));
        w.T(cutTail ?? "xref\n3 1\n" + Row(w.Offsets[3]));
        return w.Bytes;
    }

    // The first section, which startxref names, cut by the end of the file; an older table (unreachable) holds object 4.
    internal static byte[] Cut207First(string rowsOf)
    {
        var w = new W();
        w.Obj(1, "<< /Type /Catalog /Pages 2 0 R >>");
        w.Obj(2, "<< /Type /Pages /Kids [] /Count 0 >>");
        w.Obj(4, "(object 4)");
        w.T("xref\n0 3\n" + Row(0, "f", 65535) + Row(w.Offsets[1]) + Row(w.Offsets[2]) + "4 1\n" + Row(w.Offsets[4]) + "trailer\n<< /Size 5 /Root 1 0 R >>\n");
        w.Obj(3, "(object 3)");
        var head = $"startxref\n";
        var at = w.P;
        string Tail(long x) => $"startxref\n{x}\n%%EOF\n";
        long table = at + Tail(0).Length;
        while (at + Tail(table).Length != table) table = at + Tail(table).Length;
        w.T(Tail(table));
        w.T(rowsOf switch
        {
            "catalog" => "xref\n0 4\n" + Row(0, "f", 65535) + Row(w.Offsets[1]) + Row(w.Offsets[2]) + Row(w.Offsets[3]),
            "three" => "xref\n3 1\n" + Row(w.Offsets[3]),
            "none" => "xref\n",
            _ => throw new ArgumentException(rowsOf),
        });
        return w.Bytes;
    }

    // The only section, which startxref names, cut by the end of the file after the catalog's row: no other trailer.
    internal static byte[] Cut207Alone()
    {
        var w = new W();
        w.Obj(1, "<< /Type /Catalog /Pages 2 0 R >>");
        w.Obj(2, "<< /Type /Pages /Kids [] /Count 0 >>");
        w.Obj(3, "(object 3)");
        w.Obj(4, "(object 4)");
        var at = w.P;

    }

    // ---- #210: a section read only in part ----

    // A /Prev table whose last row, object 4's, ends in x, then its trailer; 1,200 bytes on, the newest table. With
    // first = true, the table is the only section, startxref naming it.
    internal static byte[] RowFault210(bool first, bool middle = false, bool secondSubsection = false)
    {
        var w = new W();
        w.Obj(1, "<< /Type /Catalog /Pages 2 0 R >>");
        w.Obj(2, "<< /Type /Pages /Kids [] /Count 0 >>");
        w.Obj(3, "(object 3)");
        w.Obj(4, "(object 4)");
        var table = w.P;
        if (middle)
        {
            w.T("xref\n0 5\n" + Row(0, "f", 65535) + Row(w.Offsets[1]) + Row(w.Offsets[2], "x") + Row(w.Offsets[3]) + Row(w.Offsets[4]) + "trailer\n<< /Size 5 /Root 1 0 R >>\n");
        }
        else if (secondSubsection)
        {
            w.T("xref\n0 3\n" + Row(0, "f", 65535) + Row(w.Offsets[1]) + Row(w.Offsets[2], "x") + "3 2\n" + Row(w.Offsets[3]) + Row(w.Offsets[4]) + "trailer\n<< /Size 5 /Root 1 0 R >>\n");
        }
        else
        {
            w.T("xref\n0 5\n" + Row(0, "f", 65535) + Row(w.Offsets[1]) + Row(w.Offsets[2]) + Row(w.Offsets[3]) + Row(w.Offsets[4], "x") + "trailer\n<< /Size 5 /Root 1 0 R >>\n");
        }

        if (first)
        {
            w.T($"startxref\n{table}\n%%EOF\n");
            return w.Bytes;
        }

        w.T($"startxref\n{table}\n%%EOF\n");
        w.Pad(1200);
        var newest = w.P;
        w.T("xref\n0 1\n" + Row(0, "f", 65535) + $"trailer\n<< /Size 5 /Root 1 0 R /Prev {table} >>\nstartxref\n{newest}\n%%EOF\n");
        return w.Bytes;
    }

    // A hybrid file whose /XRefStm stream holds one row where its /Index [4 2] declares two: object 5 is lost.
    internal static byte[] Hybrid210()
    {
        var w = new W("1.5");
        w.Obj(1, "<< /Type /Catalog /Pages 2 0 R >>");
        w.Obj(2, "<< /Type /Pages /Kids [] /Count 0 >>");
        w.Obj(3, "(object 3)");
        w.Obj(4, "(object 4)");
        w.Obj(5, "(object 5)");
        var stream = w.Stream(6, "/Type /XRef /Size 7 /W [1 4 2] /Index [4 2]", SRows([1, 4, 2], (1, (ulong)w.Offsets[4], 0, -1)));
        var table = w.P;
        w.T("xref\n0 4\n" + Row(0, "f", 65535) + Row(w.Offsets[1]) + Row(w.Offsets[2]) + Row(w.Offsets[3]) +
            $"trailer\n<< /Size 7 /Root 1 0 R /XRefStm {stream} >>\nstartxref\n{table}\n%%EOF\n");
        return w.Bytes;
    }

    // A cross-reference stream whose data holds 4 rows where /Size 6 declares 6: objects 4 and 5 lost. first: the only
    // section; otherwise named by /Prev from a newer stream that indexes itself alone.
    internal static byte[] Short210(bool first)
    {
        var w = new W("1.5");
        w.Obj(1, "<< /Type /Catalog /Pages 2 0 R >>");
        w.Obj(2, "<< /Type /Pages /Kids [] /Count 0 >>");
        w.Obj(3, "(object 3)");
        w.Obj(4, "(object 4)");
        var stream = w.P;
        w.Stream(5, "/Type /XRef /Size 6 /W [1 4 2] /Root 1 0 R", SRows([1, 4, 2], (0, 0, 65535, -1), (1, (ulong)w.Offsets[1], 0, -1), (1, (ulong)w.Offsets[2], 0, -1), (1, (ulong)w.Offsets[3], 0, -1)));
        w.T($"startxref\n{stream}\n%%EOF\n");
        if (first)
        {
            return w.Bytes;
        }

        w.Pad(1200);
        var newer = w.P;
        w.Stream(6, $"/Type /XRef /Size 7 /W [1 4 2] /Index [6 1] /Root 1 0 R /Prev {stream}", SRows([1, 4, 2], (1, (ulong)newer, 0, -1)));
        w.T($"startxref\n{newer}\n%%EOF\n");
        return w.Bytes;
    }

    private static (W W, long Table) Sound()
    {
        var w = new W();
        w.Obj(1, "<< /Type /Catalog /Pages 2 0 R >>");
        w.Obj(2, "<< /Type /Pages /Kids [3 0 R] /Count 1 >>");
        w.Obj(3, "<< /Type /Page /Parent 2 0 R /MediaBox [0 0 612 792] /Resources << >> >>");
        w.Obj(4, "(object 4)");
        var table = w.P;
        w.T("xref\n0 5\n" + Row(0, "f", 65535) + Row(w.Offsets[1]) + Row(w.Offsets[2]) + Row(w.Offsets[3]) + Row(w.Offsets[4]) +
            $"trailer\n<< /Size 5 /Root 1 0 R >>\nstartxref\n{table}\n%%EOF\n");
        return (w, table);
    }

    internal static byte[] Empty222(string shape)
    {
        if (shape == "EC2")
        {
            var a = new W();
            a.Obj(1, "<< /Type /Catalog /Pages 2 0 R >>");
            a.Obj(2, "<< /Type /Pages /Kids [3 0 R] /Count 1 >>");
            a.Obj(3, "<< /Type /Page /Parent 2 0 R /MediaBox [0 0 612 792] /Resources << >> >>");
            var t = a.P;
            a.T($"xref\ntrailer\n<< /Size 4 /Root 1 0 R >>\nstartxref\n{t}\n%%EOF\n");
            return a.Bytes;
        }

        var (w, table) = Sound();
        if (shape == "EC0")
        {
            return w.Bytes;
        }

        var update = w.P;
        var subsections = shape switch { "EU1" => "0 0\n", _ => string.Empty };
        w.T($"xref\n{subsections}trailer\n<< /Size 5 /Root 1 0 R /Prev {table} >>\nstartxref\n{update}\n%%EOF\n");
        if (shape == "EU2p")
        {
            var newest = w.P;
            w.T("xref\n0 1\n" + Row(0, "f", 65535) + $"trailer\n<< /Size 5 /Root 1 0 R /Prev {update} >>\nstartxref\n{newest}\n%%EOF\n");
        }

        return w.Bytes;
    }

    // ---- refused rows (batch 5, #210's first comment) ----

    // A classic row giving an object in use a generation outside 0..65535: the page's (object 3) in the first section, or
    // object 4's in a /Prev section under an empty update.
    internal static byte[] RefusedClassic(long generation, bool prev)
    {
        var w = new W();
        w.Obj(1, "<< /Type /Catalog /Pages 2 0 R >>");
        w.Obj(2, "<< /Type /Pages /Kids [3 0 R] /Count 1 >>");
        w.Obj(3, "<< /Type /Page /Parent 2 0 R /MediaBox [0 0 612 792] /Resources << >> >>");
        w.Obj(4, "(object 4)");
        var table = w.P;
        var bad = string.Create(CultureInfo.InvariantCulture, $"{(prev ? w.Offsets[4] : w.Offsets[3])} {generation} n").PadRight(18) + "\r\n";
        w.T("xref\n0 5\n" + Row(0, "f", 65535) + Row(w.Offsets[1]) + Row(w.Offsets[2]) + (prev ? Row(w.Offsets[3]) + bad : bad + Row(w.Offsets[4])) +
            $"trailer\n<< /Size 5 /Root 1 0 R >>\nstartxref\n{table}\n%%EOF\n");
        if (!prev)
        {
            return w.Bytes;
        }

        w.Pad(1200);
        var update = w.P;
        w.T("xref\n0 1\n" + Row(0, "f", 65535) + $"trailer\n<< /Size 5 /Root 1 0 R /Prev {table} >>\nstartxref\n{update}\n%%EOF\n");
        return w.Bytes;
    }

    // A cross-reference stream (object 5, the only section) whose row for object `number` is given (type, second, third);
    // object 4 is a string, object 3 the page, both written directly.
    internal static byte[] RefusedStream(int[] widths, int number, ulong type, ulong? second, ulong third, int lead = -1)
    {
        var w = new W("1.5");
        w.Obj(1, "<< /Type /Catalog /Pages 2 0 R >>");
        w.Obj(2, "<< /Type /Pages /Kids [3 0 R] /Count 1 >>");
        w.Obj(3, "<< /Type /Page /Parent 2 0 R /MediaBox [0 0 612 792] /Resources << >> >>");
        w.Obj(4, "(object 4)");
        var xref = w.P;
        var rows = new (ulong T, ulong S, ulong G, int Lead)[]
        {
            (0, 0, 65535, -1),
            (1, (ulong)w.Offsets[1], 0, -1),
            (1, (ulong)w.Offsets[2], 0, -1),
            (1, (ulong)w.Offsets[3], 0, -1),
            (1, (ulong)w.Offsets[4], 0, -1),
            (1, (ulong)xref, 0, -1),
        };
        rows[number] = (type, second ?? rows[number].S, third, lead);
        w.Stream(5, $"/Type /XRef /Size 6 /W [{widths[0]} {widths[1]} {widths[2]}] /Root 1 0 R", SRows(widths, rows));
        w.T($"startxref\n{xref}\n%%EOF\n");
        return w.Bytes;
    }

    // ---- neighbor: a /Prev table whose trailer keyword is followed by no dictionary ----
    internal static byte[] TrailerNotADictionary()
    {
        var w = new W();
        w.Obj(1, "<< /Type /Catalog /Pages 2 0 R >>");
        w.Obj(2, "<< /Type /Pages /Kids [] /Count 0 >>");
        w.Obj(4, "(object 4)");
        var oldest = w.P;
        w.T("xref\n4 1\n" + Row(w.Offsets[4]) + "trailer\n<< /Size 5 >>\n");
        w.Obj(3, "(object 3)");
        var middle = w.P;
        w.T("xref\n3 1\n" + Row(w.Offsets[3]) + "trailer\n[ /Size 5 /Prev " + oldest + " ]\n");
        w.Pad(1200);
        var newest = w.P;
        w.T("xref\n0 3\n" + Row(0, "f", 65535) + Row(w.Offsets[1]) + Row(w.Offsets[2]) + $"trailer\n<< /Size 5 /Root 1 0 R /Prev {middle} >>\nstartxref\n{newest}\n%%EOF\n");
        return w.Bytes;
    }

    [Fact]
    public void ZzPartialShapesAll()
    {
        Out.Clear();
        Dump("207-P: the issue's shape, a /Prev table cut by the end of the file after object 3's row", Cut207Prev(), [3, 4]);
        Dump("207-P-far: same, the update 1,200 bytes past object 3", Cut207Prev(gap: 1200), [3, 4]);
        Dump("207-P0: a /Prev table that is 'xref' alone at the end of the file", Cut207Prev("xref\n"), [3, 4]);
        Dump("207-P0-far: same, the update 1,200 bytes past object 3", Cut207Prev("xref\n", gap: 1200), [3, 4]);
        Dump("207-Prow: a /Prev table cut inside its first row", Cut207Prev("xref\n3 1\n00000001"), [3, 4]);
        Dump("207-F: the first section cut by the end of the file, holding the catalog's row; an older table holds object 4", Cut207First("catalog"), [3, 4]);
        Dump("207-F3: the first section cut, holding object 3's row alone", Cut207First("three"), [3, 4]);
        Dump("207-F0: the first section 'xref' alone at the end of the file", Cut207First("none"), [3, 4]);
        Dump("207-F1: the only section, cut by the end of the file after the catalog's row, no other trailer", Cut207Alone(), [3, 4]);

        Dump("210-P: a /Prev table whose last row, object 4's, ends in x, then its trailer; the newest table 1,200 bytes on", RowFault210(first: false), [3, 4]);
        Dump("210-X: an /XRefStm stream holding one row where its /Index [4 2] declares two", Hybrid210(), [4, 5]);
        Dump("210-F: the first (only) table, its last row ending in x, then its trailer", RowFault210(first: true), [3, 4]);
        Dump("210-FS: the first (only) section, a stream holding 4 rows where /Size declares 6", Short210(first: true), [3, 4]);
        Dump("210-PS: a /Prev stream holding 4 rows where /Size declares 6", Short210(first: false), [3, 4]);
        Dump("210-M (control): a /Prev table whose row for object 2, mid-subsection, ends in x", RowFault210(first: false, middle: true), [2, 3, 4]);
        Dump("210-2S: a /Prev table whose first subsection's last row (object 2) ends in x, a second subsection after it", RowFault210(first: false, secondSubsection: true), [2, 3, 4]);

        Dump("222-EC0 (control): the sound file alone", Empty222("EC0"), [3, 4]);
        Dump("222-EU1 (control): an update 'xref / 0 0 / trailer'", Empty222("EU1"), [3, 4]);
        Dump("222-EU2: an update 'xref / trailer', no subsection", Empty222("EU2"), [3, 4]);
        Dump("222-EU2p: the update 'xref / trailer' named by /Prev from a newer one", Empty222("EU2p"), [3, 4]);
        Dump("222-EC2 (#218's): 'xref / trailer' the only section", Empty222("EC2"), [3]);

        Dump("R-C1: classic row giving the page (object 3) generation 65536, the only section", RefusedClassic(65536, prev: false), [3, 4]);
        Dump("R-C1b: classic row giving the page generation 4294967296, the only section", RefusedClassic(4294967296, prev: false), [3, 4]);
        Dump("R-C1c: classic row giving the page generation -1, the only section", RefusedClassic(-1, prev: false), [3, 4]);
        Dump("R-C1p: classic row giving object 4 generation 65536, in a /Prev section", RefusedClassic(65536, prev: true), [3, 4]);
        Dump("R-S1: stream row giving object 4 offset 2^63 (W [1 8 2])", RefusedStream([1, 8, 2], 4, 1, 9223372036854775808UL, 0), [3, 4]);
        Dump("R-S2: stream row giving object 4 generation 65536 (W [1 4 4])", RefusedStream([1, 4, 4], 4, 1, null, 65536), [3, 4]);
        Dump("R-S3: stream row placing object 3 in object stream 0 (W [1 4 2])", RefusedStream([1, 4, 2], 3, 2, 0, 1), [3, 4]);
        Dump("R-S3b: stream row placing object 3 in object stream 2147483648 (W [1 4 2])", RefusedStream([1, 4, 2], 3, 2, 2147483648UL, 1), [3, 4]);
        Dump("R-S4: stream row placing object 3 at index 2147483648 of object stream 4 (W [1 4 8])", RefusedStream([1, 4, 8], 3, 2, 4, 2147483648UL), [3, 4]);
        Dump("R-S5: stream row giving object 4 a generation of more than 64 bits (W [1 4 9])", RefusedStream([1, 4, 9], 4, 1, null, 0, lead: 2), [3, 4]);

        Dump("N-TND (neighbor): a /Prev table whose trailer keyword is followed by an array, its /Prev lost", TrailerNotADictionary(), [3, 4]);
        Dump("N-DUP (neighbor, on main too): the only table, its trailer without /Root, the page's (object 3) row refused (generation 65536)", Encoding.Latin1.GetBytes(Encoding.Latin1.GetString(RefusedClassic(65536, prev: false)).Replace("/Size 5 /Root 1 0 R", "/Size 5 /Rot 1 0 R")), [3, 4]);

        File.WriteAllText(Environment.GetEnvironmentVariable("ZZ_OUT") ?? "/tmp/zz-partial.txt", Out.ToString());
    }
