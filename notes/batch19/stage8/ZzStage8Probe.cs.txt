using System.Diagnostics;
using System.Globalization;
using System.Text;
using AdCodicem.Pdf.Documents;
using AdCodicem.Pdf.IO.XRef;
using AdCodicem.Pdf.Objects;
using AdCodicem.Pdf.Validation;

namespace AdCodicem.Pdf.Tests;

#pragma warning disable

public class ZzStage8Probe
{
    [Fact]
    public void ZzMany()
    {
        var output = new StringBuilder();

        foreach (var every in new[] { true, false })
        {
            var bytes = HybridFiles.TablesNamingOneStream(1000, 100_000, every);
            using var source = new StrictCountingSource(bytes);
            using (var document = PdfDocument.Open(source, options: null, ownsSource: false))
            {
                var sections = document.Reader.Structure.Sections;
                output.Append(CultureInfo.InvariantCulture, $"every={every} file={bytes.Length:N0} read={source.BytesRead:N0} records={sections.Count} again={sections.Count(s => s.State == XRefSectionState.NamedAgain)} streams={sections.Count(s => s.IsHybridStream)} diags={document.Diagnostics.Count}\n");
            }

            var allocated = Allocations.Least(() => { using var d = PdfDocument.Open(bytes); return d.Reader.Structure.Sections.Count; }, out _);
            var watch = Stopwatch.StartNew();
            for (var i = 0; i < 5; i++) { using var d = PdfDocument.Open(bytes); }
            output.Append(CultureInfo.InvariantCulture, $"   allocated={allocated:N0} time5={watch.Elapsed.TotalMilliseconds:F1}ms\n");
        }

        foreach (var (name, file) in Shapes())
        {
            using var document = PdfDocument.Open(file);
            output.Append("== ").Append(name).Append('\n');
            foreach (var s in document.Reader.Structure.Sections)
            {
                output.Append(CultureInfo.InvariantCulture, $"  S {s.NamedBy} {s.State} off={s.Offset} kind={s.Kind} cut={s.CutByLimit} nns={s.NamesNoStream}\n");
            }

            foreach (var d in document.Diagnostics) output.Append("  D ").Append(d).Append('\n');
            var report = new PdfValidator().Validate(document);
            foreach (var d in document.Diagnostics) output.Append("  D' ").Append(d).Append('\n');
            foreach (var f in report.Findings) output.Append("  F ").Append(f).Append('\n');
        }

        File.WriteAllText("/tmp/claude-0/-home-user-AdCodicem-Pdf/b5ddd9fc-0507-5e11-8a94-c9e1fedd4860/scratchpad/lot19/stage8/probe.txt", output.ToString());
    }

    [Fact]
    public void ZzAlloc()
    {
        var output = new StringBuilder();
        foreach (var tables in new[] { 250, 1000 })
        {
            var every = HybridFiles.TablesNamingOneStream(tables, 100_000, true);
            var newest = HybridFiles.TablesNamingOneStream(tables, 100_000, false);
            long Read(byte[] f) { using var s = new StrictCountingSource(f); using var d = PdfDocument.Open(s, options: null, ownsSource: false); return s.BytesRead; }
            var a = Allocations.Least(() => { using var d = PdfDocument.Open(every); return 0; }, out _);
            var b = Allocations.Least(() => { using var d = PdfDocument.Open(newest); return 0; }, out _);
            var w = Stopwatch.StartNew(); for (var i = 0; i < 5; i++) { using var d = PdfDocument.Open(every); } var te = w.Elapsed.TotalMilliseconds;
            w.Restart(); for (var i = 0; i < 5; i++) { using var d = PdfDocument.Open(newest); } var tn = w.Elapsed.TotalMilliseconds;
            output.Append(CultureInfo.InvariantCulture, $"tables={tables} readEvery={Read(every):N0} readNewest={Read(newest):N0} allocEvery={a:N0} allocNewest={b:N0} diff={a - b:N0} perNaming={(a - b) / (double)(tables - 1):F1} time5 every={te:F1} newest={tn:F1}\n");
        }

        File.WriteAllText("/tmp/claude-0/-home-user-AdCodicem-Pdf/b5ddd9fc-0507-5e11-8a94-c9e1fedd4860/scratchpad/lot19/stage8/probe-alloc.txt", output.ToString());
    }

    [Fact]
    public void ZzGuarded()
    {
        var output = new StringBuilder();
        var file = HybridFiles.NamingATableNeverRead();
        foreach (var (name, limits) in new (string, PdfReaderLimits)[]
        {
            ("whole", PdfReaderLimits.Default),
            ("index 2", PdfReaderLimits.Default with { MaxIndexEntries = 2 }),
            ("index 3", PdfReaderLimits.Default with { MaxIndexEntries = 3 }),
            ("section 40", PdfReaderLimits.Default with { MaxXRefSectionLength = 40 }),
            ("section 45", PdfReaderLimits.Default with { MaxXRefSectionLength = 45 }),
            ("section 30", PdfReaderLimits.Default with { MaxXRefSectionLength = 30 }),
        })
        {
            try
            {
                using var document = PdfDocument.Open(file, new PdfReaderOptions { Limits = limits });
                output.Append("== ").Append(name).Append('\n');
                foreach (var s in document.Reader.Structure.Sections)
                {
                    output.Append(CultureInfo.InvariantCulture, $"  S {s.NamedBy} {s.State} off={s.Offset} kind={s.Kind} cut={s.CutByLimit} nns={s.NamesNoStream}\n");
                }

                foreach (var d in document.Diagnostics) output.Append("  D ").Append(d).Append('\n');
                output.Append("  ten=").Append(document.GetObject(new PdfObjectId(10))).Append(" repaired=").Append(document.WasRepaired).Append('\n');
            }
            catch (Exception e) { output.Append("  THREW ").Append(e.Message).Append('\n'); }
        }

        File.WriteAllText("/tmp/claude-0/-home-user-AdCodicem-Pdf/b5ddd9fc-0507-5e11-8a94-c9e1fedd4860/scratchpad/lot19/stage8/probe-guard.txt", output.ToString());
    }

    private static IEnumerable<(string, byte[])> Shapes()
    {
        yield return ("never-read", HybridFiles.NamingATableNeverRead());
        yield return ("prev table", HybridFiles.NamingThePreviousTable());
        yield return ("prev stream", HybridFiles.NamingThePreviousStream().File);
        yield return ("newer stream", HybridFiles.NamingANewerStream().File);
        yield return ("itself", HybridFiles.NamingItself());
        yield return ("shared", HybridFiles.SharedStream(7).File);
        yield return ("shared short", HybridFiles.SharedStreamNamedShort().File);
    }
}
